{缠论主图：笔、线段、中枢、当下买卖点（信号当下可知，结构随数据更新）。预设0笔、1线段、2所属线段中枢与快速信号}
XC:=TDXDLL1(40,C,V,0);          { 注册真实收盘价/成交量：MACD 背驰改用真实收盘价 }
BI:=TDXDLL1(1,H,L,0);           { 笔端点：顶 1 / 底 -1 }
SEG:=TDXDLL1(1,H,L,1);       { 线段端点：特征序列法 }
ZG:=TDXDLL1(2,H,L,2);           { 笔中枢上沿 }
ZD:=TDXDLL1(3,H,L,2);           { 笔中枢下沿 }
BS:=TDXDLL1(41,H,L,2);           { 快速买卖点：分型确认即提示，后续可失效 }
ST:=TDXDLL1(43,H,L,2);          { 买卖点端点分型极值失效价 }

{ 笔（黄细线）与线段（红粗线） }
DRAWLINE(BI=-1,L,BI=1,H,0),COLORYELLOW,LINETHICK1;
DRAWLINE(BI=1,H,BI=-1,L,0),COLORYELLOW,LINETHICK1;
DRAWLINE(SEG=-1,L,SEG=1,H,0),COLORRED,LINETHICK2;
DRAWLINE(SEG=1,H,SEG=-1,L,0),COLORRED,LINETHICK2;

{ 中枢区间 }
中枢上沿:IF(ZG>0,ZG,DRAWNULL),COLORMAGENTA,LINETHICK1;
中枢下沿:IF(ZD>0,ZD,DRAWNULL),COLORMAGENTA,LINETHICK1;

{ 当下买卖点（K线下方/上方）与止损点 }
DRAWTEXT(BS=1,L,'一买'),COLORRED;
DRAWTEXT(BS=2,L,'二买'),COLORRED;
DRAWTEXT(BS=3,L,'三买'),COLORRED;
DRAWTEXT(BS=11,H,'一卖'),COLORGREEN;
DRAWTEXT(BS=12,H,'二卖'),COLORGREEN;
DRAWTEXT(BS=13,H,'三卖'),COLORGREEN;


买点失效价:IF(BS>=1 AND BS<=3,ST,DRAWNULL),POINTDOT,COLORRED;
卖点失效价:IF(BS>=11 AND BS<=13,ST,DRAWNULL),POINTDOT,COLORGREEN;
