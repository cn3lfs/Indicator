{缠论主图：笔、线段、中枢、当下买卖点（无未来函数）。配置码 0=严格笔；线段/线段中枢用 1100}
XC:=TDXDLL1(40,C,V,0);          { 注册真实收盘价/成交量：MACD 背驰改用真实收盘价 }
BI:=TDXDLL1(1,H,L,0);           { 笔端点：顶 1 / 底 -1 }
SEG:=TDXDLL1(1,H,L,1100);       { 线段端点：特征序列法 }
ZG:=TDXDLL1(2,H,L,0);           { 笔中枢上沿 }
ZD:=TDXDLL1(3,H,L,0);           { 笔中枢下沿 }
BS:=TDXDLL1(41,H,L,0);           { 快速候选买卖点：分型确认即提示，后续可失效 }
ST:=TDXDLL1(43,H,L,0);          { 候选提示时的失效价 }
XX:=TDXDLL1(42,H,L,0);           { 此前出现过、在此被否定的信号 }

{ 笔（黄细线）与线段（红粗线） }
DRAWLINE(BI=-1,L,BI=1,H,0),COLORYELLOW,LINETHICK1;
DRAWLINE(BI=1,H,BI=-1,L,0),COLORYELLOW,LINETHICK1;
DRAWLINE(SEG=-1,L,SEG=1,H,0),COLORRED,LINETHICK2;
DRAWLINE(SEG=1,H,SEG=-1,L,0),COLORRED,LINETHICK2;

{ 中枢区间 }
中枢上沿:IF(ZG>0,ZG,DRAWNULL),COLORMAGENTA,LINETHICK1;
中枢下沿:IF(ZD>0,ZD,DRAWNULL),COLORMAGENTA,LINETHICK1;

{ 当下买卖点（K线下方/上方）与失效标记 }
DRAWTEXT(BS=1,L,'一买候选'),COLORRED;
DRAWTEXT(BS=2,L,'二买候选'),COLORRED;
DRAWTEXT(BS=3,L,'三买候选'),COLORRED;
DRAWTEXT(BS=11,H,'一卖候选'),COLORGREEN;
DRAWTEXT(BS=12,H,'二卖候选'),COLORGREEN;
DRAWTEXT(BS=13,H,'三卖候选'),COLORGREEN;
DRAWTEXT(XX>0,H,'失效'),COLORGRAY;

买点失效价:IF(BS>=1 AND BS<=3,ST,DRAWNULL),POINTDOT,COLORRED;
卖点失效价:IF(BS>=11 AND BS<=13,ST,DRAWNULL),POINTDOT,COLORGREEN;
